cmake_minimum_required(VERSION 3.16)
project(cmr_mission LANGUAGES CXX)
include(CTest)
add_library(cmr_mission INTERFACE)
target_compile_features(cmr_mission INTERFACE cxx_std_17)
target_include_directories(cmr_mission INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(BUILD_TESTING)
  add_executable(admission_test test/admission_test.cpp)
  target_link_libraries(admission_test PRIVATE cmr_mission)
  set_target_properties(admission_test PROPERTIES CXX_EXTENSIONS OFF)
  target_compile_options(admission_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  foreach(case validation identity duplicate conflict busy context immutable permutations)
    add_test(NAME mission_admission_${case} COMMAND admission_test ${case})
  endforeach()
endif()
# First offline slice only; ROS packaging/install/export belongs to the next steps.

add_library(mission_logic STATIC src/mission.cpp)
target_link_libraries(mission_logic PUBLIC cmr_mission)
target_compile_options(mission_logic PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
 add_executable(mission_test test/mission_test.cpp)
 target_link_libraries(mission_test PRIVATE mission_logic)
 add_test(NAME mission_state_timelines COMMAND mission_test)
endif()

find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
add_library(mission_store STATIC src/mission_store.cpp)
target_link_libraries(mission_store PUBLIC mission_logic SQLite::SQLite3 Threads::Threads)
target_compile_options(mission_store PRIVATE -Wall -Wextra -Wpedantic)

if(BUILD_TESTING)
 add_executable(store_test test/store_test.cpp)
 target_link_libraries(store_test PRIVATE mission_store)
 add_test(NAME mission_store_contract COMMAND store_test ${CMAKE_BINARY_DIR}/store-artifacts)
endif()

add_executable(mission_probe test/mission_probe.cpp)
target_link_libraries(mission_probe PRIVATE mission_store)
if(BUILD_TESTING)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME mission_crash_windows COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/crash_test.py $<TARGET_FILE:mission_probe> ${CMAKE_BINARY_DIR}/crash-artifacts)
 set_tests_properties(mission_crash_windows PROPERTIES TIMEOUT 180)
endif()

if(BUILD_TESTING)
 add_executable(interleaving_test test/interleaving_test.cpp)
 target_link_libraries(interleaving_test PRIVATE mission_store)
 add_test(NAME mission_interleaving COMMAND interleaving_test ${CMAKE_BINARY_DIR}/interleaving-artifacts)
 set_tests_properties(mission_interleaving PROPERTIES TIMEOUT 180)
endif()

if(BUILD_TESTING)
 add_executable(runtime_race_test test/runtime_race_test.cpp)
 target_link_libraries(runtime_race_test PRIVATE mission_store)
 add_test(NAME mission_runtime_races COMMAND runtime_race_test ${CMAKE_BINARY_DIR}/runtime-race-artifacts)
 set_tests_properties(mission_runtime_races PROPERTIES TIMEOUT 60)
endif()
