cmake_minimum_required(VERSION 3.16)
project(cmr_mission LANGUAGES CXX)
include(CTest)
option(CMR_MISSION_BUILD_ROS "Build ROS mission adapter and isolated test server" OFF)
add_library(cmr_mission INTERFACE)
target_compile_features(cmr_mission INTERFACE cxx_std_17)
target_include_directories(cmr_mission INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(BUILD_TESTING)
  add_executable(admission_test test/admission_test.cpp)
  target_link_libraries(admission_test PRIVATE cmr_mission)
  set_target_properties(admission_test PROPERTIES CXX_EXTENSIONS OFF)
  target_compile_options(admission_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  foreach(case validation identity duplicate conflict busy context immutable permutations)
    add_test(NAME mission_admission_${case} COMMAND admission_test ${case})
  endforeach()
endif()
# Core targets remain usable without ROS; optional ament packaging is below.

add_library(mission_logic STATIC src/mission.cpp)
target_link_libraries(mission_logic PUBLIC cmr_mission)
target_compile_options(mission_logic PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
 add_executable(mission_test test/mission_test.cpp)
 target_link_libraries(mission_test PRIVATE mission_logic)
 add_test(NAME mission_state_timelines COMMAND mission_test)
endif()
