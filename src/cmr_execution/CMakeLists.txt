cmake_minimum_required(VERSION 3.16)
project(cmr_execution LANGUAGES CXX)
include(CTest)
option(CMR_EXECUTION_BUILD_ROS "Build independent ROS execution Gate" OFF)
add_library(execution_core STATIC src/core.cpp)
target_compile_features(execution_core PUBLIC cxx_std_17)
target_include_directories(execution_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/../cmr_core/include> $<INSTALL_INTERFACE:include>)
target_compile_options(execution_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(BUILD_TESTING)
 find_package(Threads REQUIRED)
 add_executable(execution_test test/execution_test.cpp)
 target_link_libraries(execution_test PRIVATE execution_core Threads::Threads)
 target_compile_options(execution_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
 foreach(case body identity normal idempotence freshness clock restart boundaries concurrency stop_good stop_bad reset unissued_cancel time_format random)
  add_test(NAME execution_${case} COMMAND execution_test ${case})
 endforeach()
endif()
if(CMR_EXECUTION_BUILD_ROS)
 find_package(ament_cmake REQUIRED)
 find_package(rclcpp REQUIRED)
 find_package(cmr_interfaces REQUIRED)
 find_package(geometry_msgs REQUIRED)
 find_package(nav_msgs REQUIRED)
 find_package(rosgraph_msgs REQUIRED)
 find_package(cmr_core REQUIRED)
 target_link_libraries(execution_core PUBLIC cmr_core::cmr_core)
 add_executable(candidate_relay src/candidate_relay.cpp)
 ament_target_dependencies(candidate_relay rclcpp geometry_msgs)
 install(DIRECTORY launch/ DESTINATION share/${PROJECT_NAME}/launch)
 add_executable(execution_gate src/gate_node.cpp)
 target_link_libraries(execution_gate execution_core)
 ament_target_dependencies(execution_gate rclcpp cmr_interfaces geometry_msgs rosgraph_msgs)
 add_executable(execution_observer src/stop_observer.cpp)
 target_link_libraries(execution_observer execution_core)
 ament_target_dependencies(execution_observer rclcpp cmr_interfaces geometry_msgs nav_msgs rosgraph_msgs)
 if(BUILD_TESTING)
  install(TARGETS execution_test DESTINATION lib/${PROJECT_NAME})
 endif()
 install(TARGETS candidate_relay execution_gate execution_observer DESTINATION lib/${PROJECT_NAME})
 install(DIRECTORY include/ DESTINATION include)
 install(TARGETS execution_core EXPORT execution_core_targets ARCHIVE DESTINATION lib)
 ament_export_targets(execution_core_targets HAS_LIBRARY_TARGET)
 ament_export_include_directories(include)
 ament_export_dependencies(cmr_core)
 ament_package()
endif()
