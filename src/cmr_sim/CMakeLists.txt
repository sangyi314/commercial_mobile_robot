cmake_minimum_required(VERSION 3.16)
project(cmr_sim LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz_msgs_vendor REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(cmr_scene_driver src/scene_client.cpp)
add_executable(cmr_truth_capture src/scene_client.cpp)
target_compile_definitions(cmr_truth_capture PRIVATE CMR_CAPTURE=1)
target_compile_features(cmr_scene_driver PRIVATE cxx_std_17)
target_compile_features(cmr_truth_capture PRIVATE cxx_std_17)
target_compile_options(cmr_scene_driver PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(cmr_truth_capture PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(cmr_scene_driver PRIVATE include)
target_include_directories(cmr_truth_capture PRIVATE include)
target_link_libraries(cmr_scene_driver PRIVATE gz-transport13::core gz-msgs10::gz-msgs10 nlohmann_json::nlohmann_json)
target_link_libraries(cmr_truth_capture PRIVATE gz-transport13::core gz-msgs10::gz-msgs10 nlohmann_json::nlohmann_json)
enable_testing()
add_executable(scene_timing_test test/scene_timing_test.cpp)
target_compile_features(scene_timing_test PRIVATE cxx_std_17)
target_include_directories(scene_timing_test PRIVATE include)
add_test(NAME scene_timing COMMAND scene_timing_test)
install(TARGETS cmr_scene_driver cmr_truth_capture DESTINATION lib/cmr_sim)
install(DIRECTORY models worlds DESTINATION share/cmr_sim)
ament_package()
add_executable(raw_proto_test test/raw_proto_test.cpp)
target_compile_features(raw_proto_test PRIVATE cxx_std_17)
target_link_libraries(raw_proto_test PRIVATE gz-msgs10::gz-msgs10 nlohmann_json::nlohmann_json)
add_test(NAME raw_proto_roundtrip COMMAND raw_proto_test)
set_tests_properties(scene_timing raw_proto_roundtrip PROPERTIES TIMEOUT 10)
