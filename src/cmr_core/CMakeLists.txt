cmake_minimum_required(VERSION 3.16)
project(cmr_core LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
add_library(cmr_core INTERFACE)
target_compile_features(cmr_core INTERFACE cxx_std_17)
target_include_directories(cmr_core INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(command_policy_test test/command_policy_test.cpp)
  target_link_libraries(command_policy_test cmr_core)
  target_compile_options(command_policy_test PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(DIRECTORY include/ DESTINATION include)
install(TARGETS cmr_core EXPORT cmr_core_targets)
ament_export_targets(cmr_core_targets)
ament_export_include_directories(include)
ament_package()
