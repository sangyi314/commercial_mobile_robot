cmake_minimum_required(VERSION 3.16)
project(cmr_core LANGUAGES CXX)
include(CTest)
# Optional: the core and its tests also build without ROS.
find_package(ament_cmake QUIET)
add_library(cmr_core INTERFACE)
target_compile_features(cmr_core INTERFACE cxx_std_17)
target_include_directories(cmr_core INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
if(BUILD_TESTING)
  add_executable(core_test test/core_test.cpp)
  target_link_libraries(core_test PRIVATE cmr_core)
  set_target_properties(core_test PROPERTIES CXX_EXTENSIONS OFF)
  target_compile_options(core_test PRIVATE -Wall -Wextra -Wpedantic)
  foreach(case straight arc near_zero wheels gate)
    add_test(NAME ${case} COMMAND core_test ${case})
  endforeach()
endif()
install(DIRECTORY include/ DESTINATION include)
install(TARGETS cmr_core EXPORT cmr_core_targets)
if(ament_cmake_FOUND)
  ament_export_targets(cmr_core_targets)
  ament_export_include_directories(include)
  ament_package()
else()
  # Export the same header-only target and its transitive C++17 requirement.
  install(EXPORT cmr_core_targets FILE cmr_coreConfig.cmake
    NAMESPACE cmr_core:: DESTINATION lib/cmake/cmr_core)
endif()
