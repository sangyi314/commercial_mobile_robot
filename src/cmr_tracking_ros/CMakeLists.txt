cmake_minimum_required(VERSION 3.16)
project(cmr_tracking_ros LANGUAGES CXX)
include(CTest)

find_package(ament_cmake REQUIRED)
find_package(cmr_execution REQUIRED)
find_package(cmr_interfaces REQUIRED)
find_package(cmr_tracking REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_msgs REQUIRED)

add_library(p2_input_adapter src/input_adapter.cpp)
target_include_directories(p2_input_adapter PUBLIC include)
target_link_libraries(p2_input_adapter cmr_tracking::cmr_tracking)
ament_target_dependencies(p2_input_adapter cmr_interfaces nav_msgs sensor_msgs)

add_executable(sensor_health_node src/sensor_health_node.cpp)
target_link_libraries(sensor_health_node p2_input_adapter nlohmann_json::nlohmann_json)
ament_target_dependencies(sensor_health_node cmr_execution cmr_interfaces rclcpp
  rosgraph_msgs std_srvs tf2 tf2_msgs)

foreach(target p2_input_adapter sensor_health_node)
  target_compile_features(${target} PUBLIC cxx_std_17)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()

if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  add_executable(input_adapter_test test/input_adapter_test.cpp)
  target_link_libraries(input_adapter_test p2_input_adapter GTest::gtest_main)
  add_test(NAME p2_message_conversion COMMAND input_adapter_test)
endif()

install(TARGETS sensor_health_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch/ DESTINATION share/${PROJECT_NAME}/launch)
install(DIRECTORY ../cmr_bringup/config/p2/ DESTINATION share/${PROJECT_NAME}/config)
ament_package()
